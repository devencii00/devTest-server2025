## Basic Setup (Activity 1)
New-LocalUser -Name "SysAd" -FullName "System Administrator" -Description "Administrator of this Computer" -Password "P@ssw0rd01" -PasswordNeverExpires -AccountNeverExpires
Net User SysAd
Add-LocalGroupMember -Group "Administrators" -Member "SysAd"
Install-WindowsFeature -name AD-Domain-Services -IncludeManagementTools
Install-ADDSForest   (then answer the prompts)
Install-WindowsFeature -name Web-Server -IncludeManagementTools
Invoke-WebRequest http://localhost
New-SmbShare -Name "mySharedFolder" -Path "C:\inetpub\wwwroot" -FullAccess "Everyone"

## Activity 2: AD OUs, Users, and GPOs
New-ADOrganizationalUnit -Name "Admin_OU" -Path "DC=srv,DC=world"
New-ADOrganizationalUnit -Name "Staff_OU" -Path "DC=srv,DC=world"
New-ADUser -Name "AdminUser" -Path "OU=Admin_OU,DC=srv,DC=world"
New-ADUser -Name "StaffUser" -Path "OU=Staff_OU,DC=srv,DC=world"
New-GPO -Name "Admin_GPO"
New-GPO -Name "Staff_GPO"

## Activity 2: Folder Redirection & Drive Mapping
New-GPOFolderRedirection -Name "Admin_GPO" -Path "\\192.168.1.10\AdminData" -Access "Full"
New-GPOFolderRedirection -Name "Staff_GPO" -Path "\\192.168.1.10\StaffData" -Access "Read"
New-GPODriveMap -Name "Admin_GPO" -Drive "Z:" -Path "\\192.168.1.10\AdminData"
New-GPODriveMap -Name "Staff_GPO" -Drive "Y:" -Path "\\192.168.1.10\StaffData"

Extras: sconfig, hostname, ipconfig, whoami, Get-Date, Get-SmbShare, cls, exit

Client ping needs: sconfig 4 → Enable server response to Ping.
